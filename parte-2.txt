
      sm: "w-9 h-9 text-sm",
      md: "w-11 h-11 text-base",
      lg: "w-24 h-24 text-3xl",
    };

    const initials = (profile?.full_name || "N")
      .slice(0, 1)
      .toUpperCase();

    return (
      <div
        className={`${dimensions[size]} shrink-0 rounded-full overflow-hidden bg-gradient-to-br from-blue-100 to-sky-50 text-blue-600 flex items-center justify-center font-bold ring-1 ring-blue-100`}
      >
        {profile?.avatar_url ? (

  }

  if (loading) {
    return (
      <main className="min-h-screen bg-gradient-to-br from-white via-slate-50 to-blue-50 flex items-center justify-center">
        <div className="text-center">

            <Sparkles size={30} />
          </div>
          <LoaderCircle className="animate-spin text-blue-500 mx-auto" />
          <p className="text-slate-500 text-sm mt-3">Iniciando Nexora...</p>

          <div className="w-16 h-16 rounded-2xl bg-gradient-to-br from-blue-500 to-sky-400 text-white flex items-center justify-center shadow-lg shadow-blue-200 mb-6">

            type="password"

            onChange={(e) => setPassword(e.target.value)}
            placeholder="Contraseña"
            className="glass-input mb-4"
          />

          {error && (
            <p className="text-sm text-red-600 bg-red-50 p-3 rounded-xl mb-4">
              {error}

          )}

          {notice && (
            <p className="text-sm text-emerald-700 bg-emerald-50 p-3 rounded-xl mb-4">
              {notice}

          <button
            onClick={handleAuth}

          >
            {saving
              ? "Un momento..."
              : register
                ? "Crear cuenta"
                : "Iniciar sesión"}
          </button>

          <button
            onClick={() => {

          >

        <div className="absolute top-1/2 -left-40 w-96 h-96 rounded-full bg-sky-100/50 blur-3xl" />
      </div>

      <header className="sticky top-0 z-40 bg-white/70 backdrop-blur-2xl border-b border-white/80 shadow-sm">
        <div className="max-w-7xl mx-auto px-4 sm:px-6 h-[72px] flex items-center justify-between">
          <button
            onClick={() => navigate("Inicio")}
            className="flex items-center gap-2"
          >
            <div className="w-10 h-10 rounded-xl bg-gradient-to-br from-blue-500 to-sky-400 flex items-center justify-center text-white shadow-md shadow-blue-200">
              <Sparkles size={22} />
            </div>
            <span className="text-2xl font-black tracking-tight">
              Nexora<span className="text-blue-600">.</span>
            </span>
          </button>

          <div className="hidden md:flex items-center gap-2 text-sm text-slate-500">
            <span className="w-2 h-2 bg-emerald-400 rounded-full" />
            Tu espacio social
          </div>

          <div className="flex items-center gap-2">
            <button
              onClick={() => navigate("Perfil")}
              className="flex items-center gap-2 rounded-full bg-white/80 border border-white shadow-sm p-1.5 pr-3 hover:bg-white transition"
            >
              <Avatar
                profile={{

                  username,
                  full_name: name,
                  bio,
                  avatar_url: avatar,

                size="sm"

            <button

            >
              {mobileMenuOpen ? <X size={20} /> : <MenuIcon size={20} />}

        </div>

      {mobileMenuOpen && (

            return (

                key={item.label}
                onClick={() => navigate(item.label)}
                className="w-full flex items-center gap-3 p-3 rounded-xl text-left hover:bg-blue-50"
              >

                {item.label}
              </button>
            );
          })}
        </div>
      )}

      <div className="relative max-w-7xl mx-auto px-3 sm:px-6 py-6 md:py-8 grid grid-cols-1 md:grid-cols-[230px_minmax(0,1fr)] gap-6">
        <aside className="hidden md:block">
          <div className="sticky top-24 bg-white/65 backdrop-blur-2xl border border-white rounded-3xl p-3 shadow-lg shadow-slate-200/30">
            <div className="p-3 mb-2">
              <p className="text-xs font-bold uppercase tracking-widest text-slate-400">
                Menú principal
              </p>
            </div>

            {menu.map((item) => {

                <button
                  key={item.label}
                  onClick={() => navigate(item.label)}
                  className={`w-full flex items-center gap-3 px-4 py-3.5 rounded-2xl mb-1 transition ${
                    active
                      ? "bg-blue-600 text-white shadow-lg shadow-blue-200"
                      : "text-slate-500 hover:bg-white hover:text-blue-600"

            <div className="mt-5 p-4 rounded-2xl bg-gradient-to-br from-blue-50 to-white border border-blue-50">
              <Sparkles className="text-blue-500 mb-2" size={21} />

                Tu comunidad, tus ideas, tus conexiones.
              </p>
            </div>
          </div>
        </aside>

        <section className="min-w-0">
          {error && (
            <div className="mb-5 p-4 rounded-2xl bg-red-50/90 border border-red-100 text-red-700 text-sm">
              {error}
            </div>
          )}

          {notice && (
            <div className="mb-5 p-4 rounded-2xl bg-emerald-50/90 border border-emerald-100 text-emerald-700 text-sm">
              {notice}
            </div>
          )}

          {page === "Inicio" && (
            <>
              <div className="mb-7">
                <p className="text-blue-600 text-sm font-bold mb-1">
                  TU COMUNIDAD
                </p>
                <h1 className="text-3xl sm:text-4xl font-black tracking-tight">
                  Inicio
                </h1>
                <p className="text-slate-500 mt-2">
                  Esto está pasando en Nexora.
                </p>
              </div>

              <div className="glass-panel p-4 sm:p-6 mb-7">
                <div className="flex items-center gap-3 mb-4">
                  <Avatar
                    profile={{
                      id: user.id,
                      username,
                      full_name: name,
                      bio,
                      avatar_url: avatar,
                    }}
                  />
                  <div className="min-w-0">
                    <p className="font-bold truncate">{name}</p>

                <textarea
                  value={postText}
                  onChange={(e) => setPostText(e.target.value)}
                  placeholder="¿Qué quieres compartir hoy?"
                  className="w-full min-h-32 resize-y bg-white/70 border border-white rounded-2xl p-4 outline-none focus:ring-2 focus:ring-blue-100 placeholder:text-slate-400"

                    Comparte algo con tu comunidad.

                  <button
                    onClick={publish}
                    disabled={saving || !postText.trim()}
                    className="flex items-center gap-2 bg-blue-600 hover:bg-blue-700 text-white px-5 py-3 rounded-xl font-bold shadow-lg shadow-blue-200 disabled:opacity-50"
                  >

                  </button>

              </div>

              <h2 className="font-black text-xl mb-4">Tu feed</h2>

              {posts.length === 0 && (
                <div className="glass-panel p-10 text-center">
                  <div className="w-14 h-14 rounded-2xl bg-blue-50 text-blue-500 flex items-center justify-center mx-auto mb-4">
                    <Sparkles size={28} />
                  </div>
                  <p className="font-bold">Todavía no hay publicaciones.</p>
                  <p className="text-sm text-slate-500 mt-2">
                    ¡Comparte la primera!
                  </p>
                </div>
              )}

              {posts.map((post) => {
                const author = getProfile(post.user_id);
                const comments = commentsByPost[post.id] || [];
                const liked = likedPosts.includes(post.id);
                const expanded = !!expandedComments[post.id];

                return (
                  <article
                    key={post.id}
                    className="glass-panel p-4 sm:p-6 mb-4"
                  >
                    <div className="flex items-center gap-3">
                      <Avatar profile={author} />
                      <div className="min-w-0 flex-1">
                        <p className="font-bold truncate">
                          {author?.full_name || "Usuario Nexora"}
                        </p>
                        <p className="text-xs text-slate-400">
                          {author?.username
                            ? `@${author.username} · `
                            : ""}
                          {new Date(post.created_at).toLocaleString()}
                        </p>
                      </div>
                    </div>

                    <p className="my-5 whitespace-pre-wrap leading-relaxed break-words">
                      {post.content}
                    </p>

                    <div className="flex items-center gap-3 border-t border-slate-100 pt-3">
                      <button
                        onClick={() => toggleLike(post.id)}
                        className={`flex items-center gap-2 px-3 py-2 rounded-xl transition ${
                          liked
                            ? "bg-rose-50 text-rose-600"
                            : "text-slate-500 hover:bg-rose-50 hover:text-rose-500"
                        }`}
                      >
                        <Heart
                          size={19}
                          fill={liked ? "currentColor" : "none"}
                        />

                          Me gusta {likesByPost[post.id] || 0}

                      <button

                          await loadComments(post.id);

                      >
                        <MessageCircle size={19} />
                        <span className="text-sm font-semibold">
                          {comments.length} comentarios
                        </span>
                      </button>
                    </div>

                    {expanded && (

                          <input

                            onChange={(e) =>

                                ...previous,
                                [post.id]: e.target.value,
                              }))
                            }
                            onKeyDown={(e) => {

                            placeholder="Escribe un comentario..."

                          />
                          <button

                            className="bg-blue-600 text-white px-4 rounded-xl disabled:opacity-50"
                          >

                        {comments.length === 0 && (
                          <p className="text-sm text-slate-400 text-center py-3">

                        <div className="space-y-3">
                          {comments.map((comment) => {
                            const commentAuthor =

                              post.user_id === user.id;
