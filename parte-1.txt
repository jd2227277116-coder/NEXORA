type Profile = {
  id: string;
  username: string | null;
  full_name: string | null;
  bio: string | null;
  avatar_url: string | null;
};

type Post = {
  id: string;
  user_id: string;
  content: string;
  created_at: string;
};

type Message = {
  id: string;
  sender_id: string;
  recipient_id: string;
  content: string;
  created_at: string;
  read_at: string | null;
};

type Comment = {
  id: string;
  post_id: string;
  user_id: string;
  content: string;
  created_at: string;
};

type PageName =
  | "Inicio"
  | "Descubrir"
  | "Mensajes"
  | "Notificaciones"
  | "Perfil";

const menu = [

  { label: "Descubrir" as PageName, icon: Compass },

  { label: "Notificaciones" as PageName, icon: Bell },
  { label: "Perfil" as PageName, icon: UserRound },

export default function HomePage() {
  const [user, setUser] = useState<any>(null);
  const [loading, setLoading] = useState(true);
  const [register, setRegister] = useState(false);

  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [fullName, setFullName] = useState("");

  const [page, setPage] = useState<PageName>("Inicio");
  const [name, setName] = useState("");
  const [username, setUsername] = useState("");
  const [bio, setBio] = useState("");
  const [avatar, setAvatar] = useState("");

  const [posts, setPosts] = useState<Post[]>([]);
  const [postText, setPostText] = useState("");

  const [searchTerm, setSearchTerm] = useState("");
  const [searchResults, setSearchResults] = useState<Profile[]>([]);
  const [postResults, setPostResults] = useState<Post[]>([]);
  const [searching, setSearching] = useState(false);

  const [selectedContact, setSelectedContact] =
    useState<Profile | null>(null);
  const [messages, setMessages] = useState<Message[]>([]);
  const [messageText, setMessageText] = useState("");

  const [commentsByPost, setCommentsByPost] = useState<
    Record<string, Comment[]>
  >({});
  const [commentDrafts, setCommentDrafts] = useState<
    Record<string, string>
  >({});
  const [expandedComments, setExpandedComments] = useState<
    Record<string, boolean>
  >({});

  const [likesByPost, setLikesByPost] = useState<
    Record<string, number>
  >({});
  const [likedPosts, setLikedPosts] = useState<string[]>([]);

  const [profilesById, setProfilesById] = useState<
    Record<string, Profile>
  >({});

  const [error, setError] = useState("");
  const [notice, setNotice] = useState("");
  const [saving, setSaving] = useState(false);

  const [mobileMenuOpen, setMobileMenuOpen] = useState(false);

  const fileRef = useRef<HTMLInputElement>(null);
  const messagesEndRef = useRef<HTMLDivElement>(null);

  function notifyError(message: string) {
    setError(message);
    setNotice("");
    window.setTimeout(() => setError(""), 4500);
  }

  function notifySuccess(message: string) {
    setNotice(message);
    setError("");
    window.setTimeout(() => setNotice(""), 3500);
  }

  async function loadData(currentUser: any) {
    if (!currentUser) return;

    const { data: profile, error: profileError } = await supabase
      .from("profiles")
      .select("*")
      .eq("id", currentUser.id)
      .maybeSingle();

    if (profileError) {
      notifyError("No se pudo cargar tu perfil.");
    }

    if (profile) {
      setName(profile.full_name || "");
      setUsername(profile.username || "");
      setBio(profile.bio || "");
      setAvatar(profile.avatar_url || "");
    } else {
      const initialName =
        currentUser.user_metadata?.full_name ||
        currentUser.email?.split("@")[0] ||
        "Usuario";

      setName(initialName);

      await supabase.from("profiles").upsert({
        id: currentUser.id,
        full_name: initialName,
        username: `usuario_${currentUser.id.slice(0, 8)}`,
        bio: "",
      });
    }

    const { data: postData, error: postError } = await supabase
      .from("posts")
      .select("*")
      .order("created_at", { ascending: false });

    if (postError) {
      notifyError("No se pudieron cargar las publicaciones.");
    } else {
      const loadedPosts = (postData || []) as Post[];
      setPosts(loadedPosts);
      await loadPostDetails(loadedPosts, currentUser.id);
    }
  }

  async function loadPostDetails(
    loadedPosts: Post[],
    currentUserId: string
  ) {
    if (!loadedPosts.length) {
      setCommentsByPost({});
      setLikesByPost({});
      setLikedPosts([]);
      return;
    }

    const postIds = loadedPosts.map((post) => post.id);
    const authorIds = [...new Set(loadedPosts.map((post) => post.user_id))];

    const { data: profiles } = await supabase
      .from("profiles")
      .select("*")
      .in("id", authorIds);

    const profileMap: Record<string, Profile> = {};
    (profiles || []).forEach((profile: Profile) => {
      profileMap[profile.id] = profile;
    });

    setProfilesById((previous) => ({ ...previous, ...profileMap }));

    const { data: comments } = await supabase
      .from("comments")
      .select("*")
      .in("post_id", postIds)
      .order("created_at", { ascending: true });

    const commentMap: Record<string, Comment[]> = {};
    (comments || []).forEach((comment: Comment) => {
      if (!commentMap[comment.post_id]) {
        commentMap[comment.post_id] = [];
      }
      commentMap[comment.post_id].push(comment);
    });
    setCommentsByPost(commentMap);

    // La tabla likes debe tener post_id y user_id.
    const { data: likes, error: likesError } = await supabase
      .from("likes")
      .select("post_id,user_id")
      .in("post_id", postIds);

    if (!likesError && likes) {
      const counts: Record<string, number> = {};
      const mine: string[] = [];

      likes.forEach((like: { post_id: string; user_id: string }) => {
        counts[like.post_id] = (counts[like.post_id] || 0) + 1;

        if (like.user_id === currentUserId) {
          mine.push(like.post_id);
        }
      });

      setLikesByPost(counts);
      setLikedPosts(mine);
    }
  }

  useEffect(() => {
    let active = true;

    async function initialize() {
      const { data } = await supabase.auth.getSession();

      if (!active) return;

      setUser(data.session?.user || null);

      if (data.session?.user) {
        await loadData(data.session.user);
      }

      if (active) setLoading(false);
    }

    initialize();

    const {
      data: { subscription },
    } = supabase.auth.onAuthStateChange((_event, session) => {
      setUser(session?.user || null);

      if (session?.user) {
        setTimeout(() => {
          if (active) loadData(session.user);
        }, 0);
      }
    });

    return () => {
      active = false;
      subscription.unsubscribe();
    };
  }, []);

  async function handleAuth() {
    setSaving(true);
    setError("");

    try {
      if (register) {
        if (!fullName.trim()) {
          notifyError("Escribe tu nombre.");
          setSaving(false);
          return;
        }

        const { data, error: authError } = await supabase.auth.signUp({
          email,
          password,
          options: {
            data: { full_name: fullName.trim() },
          },
        });

        if (authError) throw authError;

        if (data.user && data.session) {
          await supabase.from("profiles").upsert({
            id: data.user.id,
            full_name: fullName.trim(),
            username: `usuario_${data.user.id.slice(0, 8)}`,
            bio: "",
          });

          setUser(data.user);
          await loadData(data.user);
          notifySuccess("¡Tu cuenta Nexora está lista!");
        } else {
          notifySuccess(
            "Cuenta creada. Revisa tu correo si necesitas confirmar el registro."
          );
        }
      } else {
        const { data, error: authError } =
          await supabase.auth.signInWithPassword({
            email,
            password,
          });

        if (authError) throw authError;

        setUser(data.user);
        await loadData(data.user);
        notifySuccess("Sesión iniciada correctamente.");
      }
    } catch (err: any) {
      notifyError(err.message || "No se pudo completar la operación.");
    } finally {
      setSaving(false);
    }
  }

  async function logout() {
    await supabase.auth.signOut();
    setUser(null);
    setPosts([]);
    setMessages([]);
    setSelectedContact(null);
    setPage("Inicio");
  }

  async function saveProfile() {
    if (!username.trim() || username.trim().length < 3) {
      notifyError("El usuario debe tener al menos 3 caracteres.");
      return;
    }

    if (/\s/.test(username)) {
      notifyError("El nombre de usuario no puede tener espacios.");
      return;
    }

    setSaving(true);

    const { error: saveError } = await supabase
      .from("profiles")
      .upsert({
        id: user.id,
        username: username.trim().replace(/^@/, ""),
        full_name: name.trim(),
        bio: bio.trim(),
        avatar_url: avatar || null,
      });

    setSaving(false);

    if (saveError) {
      notifyError(
        saveError.code === "23505"
          ? "Ese nombre de usuario ya está ocupado."
          : saveError.message
      );
    } else {
      notifySuccess("Perfil actualizado.");
      await loadData(user);
    }
  }

  async function uploadAvatar(file: File) {
    if (file.size > 5 * 1024 * 1024) {
      notifyError("La imagen debe pesar menos de 5 MB.");
      return;
    }

    setSaving(true);

    try {
      const extension = file.name.split(".").pop() || "jpg";
      const path = `${user.id}/${Date.now()}.${extension}`;

      const { error: uploadError } = await supabase.storage
        .from("avatars")
        .upload(path, file, { upsert: true });

      if (uploadError) throw uploadError;

      const { data } = supabase.storage
        .from("avatars")
        .getPublicUrl(path);

      setAvatar(data.publicUrl);

      await supabase.from("profiles").upsert({
        id: user.id,
        avatar_url: data.publicUrl,
        full_name: name,
        username,
        bio,
      });

      notifySuccess("Fotografía actualizada.");
    } catch (err: any) {
      notifyError(err.message || "No se pudo subir la fotografía.");
    } finally {
      setSaving(false);
    }
  }

  async function publish() {
    if (!postText.trim()) return;

    setSaving(true);

    const { error: publishError } = await supabase.from("posts").insert({
      user_id: user.id,
      content: postText.trim(),
    });

    setSaving(false);

    if (publishError) {
      notifyError("No se pudo publicar: " + publishError.message);
      return;
    }

    setPostText("");
    await loadData(user);
    notifySuccess("¡Publicación compartida!");
  }

  async function toggleLike(postId: string) {
    const alreadyLiked = likedPosts.includes(postId);

    if (alreadyLiked) {
      const { error: deleteError } = await supabase
        .from("likes")
        .delete()
        .eq("post_id", postId)
        .eq("user_id", user.id);

      if (deleteError) {
        notifyError("No se pudo quitar el Me gusta.");
        return;
      }

      setLikedPosts((previous) =>
        previous.filter((id) => id !== postId)
      );
      setLikesByPost((previous) => ({
        ...previous,
        [postId]: Math.max(0, (previous[postId] || 0) - 1),
      }));
    } else {
      const { error: insertError } = await supabase.from("likes").insert({
        post_id: postId,
        user_id: user.id,
      });

      if (insertError) {
        notifyError(
          "No se pudo dar Me gusta. Comprueba la estructura y las políticas de la tabla likes."
        );
        return;
      }

      setLikedPosts((previous) => [...previous, postId]);
      setLikesByPost((previous) => ({
        ...previous,
        [postId]: (previous[postId] || 0) + 1,
      }));
    }
  }

  async function loadComments(postId: string) {
    const { data, error: commentsError } = await supabase
      .from("comments")
      .select("*")
      .eq("post_id", postId)
      .order("created_at", { ascending: true });

    if (commentsError) {
      notifyError("No se pudieron cargar los comentarios.");
      return;
    }

    setCommentsByPost((previous) => ({
      ...previous,
      [postId]: (data || []) as Comment[],
    }));

    const authorIds = [
      ...new Set((data || []).map((comment: Comment) => comment.user_id)),
    ];

    if (authorIds.length) {
      const { data: profiles } = await supabase
        .from("profiles")
        .select("*")
        .in("id", authorIds);

      const map: Record<string, Profile> = {};
      (profiles || []).forEach((profile: Profile) => {
        map[profile.id] = profile;
      });

      setProfilesById((previous) => ({ ...previous, ...map }));
    }
  }

  async function addComment(postId: string) {
    const content = (commentDrafts[postId] || "").trim();
    if (!content) return;

    const { error: commentError } = await supabase
      .from("comments")
      .insert({
        post_id: postId,
        user_id: user.id,
        content,
      });

    if (commentError) {
      notifyError("No se pudo publicar el comentario.");
      return;
    }

    setCommentDrafts((previous) => ({ ...previous, [postId]: "" }));
    await loadComments(postId);
    notifySuccess("Comentario publicado.");
  }

  async function deleteComment(comment: Comment) {
    const post = posts.find((item) => item.id === comment.post_id);
    const canDelete =
      comment.user_id === user.id || post?.user_id === user.id;

    if (!canDelete) {
      notifyError("No puedes eliminar este comentario.");
      return;
    }

    const { error: deleteError } = await supabase
      .from("comments")
      .delete()
      .eq("id", comment.id);

    if (deleteError) {
      notifyError(
        "No se pudo eliminar. Comprueba las políticas de seguridad de Supabase."
      );
      return;
    }

    await loadComments(comment.post_id);
    notifySuccess("Comentario eliminado.");
  }

  async function searchUsers() {
    const term = searchTerm.trim();

    if (!term) {
      setSearchResults([]);
      setPostResults([]);
      return;
    }

    setSearching(true);

    const cleanTerm = term.replace(/^@/, "");

    const [{ data: people }, { data: foundPosts }] = await Promise.all([
      supabase
        .from("profiles")
        .select("*")
        .or(`full_name.ilike.%${cleanTerm}%,username.ilike.%${cleanTerm}%`)
        .neq("id", user.id)
        .limit(20),

      supabase
        .from("posts")
        .select("*")
        .ilike("content", `%${cleanTerm}%`)
        .order("created_at", { ascending: false })
        .limit(30),
    ]);

    setSearchResults((people || []) as Profile[]);
    setPostResults((foundPosts || []) as Post[]);
    setSearching(false);
  }

  async function openConversation(person: Profile) {
    setSelectedContact(person);
    setPage("Mensajes");

    const { data } = await supabase
      .from("profiles")
      .select("*")
      .eq("id", person.id)
      .maybeSingle();

    if (data) setSelectedContact(data as Profile);
  }

  async function loadMessages() {
    if (!user || !selectedContact) return;

    const { data, error: messageError } = await supabase
      .from("messages")
      .select("*")
      .or(
        `and(sender_id.eq.${user.id},recipient_id.eq.${selectedContact.id}),and(sender_id.eq.${selectedContact.id},recipient_id.eq.${user.id})`
      )
      .order("created_at", { ascending: true });

    if (!messageError) {
      setMessages((data || []) as Message[]);
    }

    await supabase
      .from("messages")
      .update({ read_at: new Date().toISOString() })
      .eq("sender_id", selectedContact.id)
      .eq("recipient_id", user.id)
      .is("read_at", null);
  }

  useEffect(() => {
    if (!user || !selectedContact) return;

    loadMessages();

    const interval = window.setInterval(loadMessages, 5000);

    return () => window.clearInterval(interval);
  }, [user, selectedContact]);

  useEffect(() => {
    messagesEndRef.current?.scrollIntoView({ behavior: "smooth" });
  }, [messages]);

  async function sendMessage() {
    const content = messageText.trim();

    if (!content || !selectedContact) return;

    const { error: sendError } = await supabase.from("messages").insert({
      sender_id: user.id,
      recipient_id: selectedContact.id,
      content,
    });

    if (sendError) {
      notifyError("No se pudo enviar el mensaje.");
      return;
    }

    setMessageText("");
    await loadMessages();
  }

  function navigate(destination: PageName) {
    setPage(destination);
    setMobileMenuOpen(false);

    if (destination !== "Mensajes") {
      setSelectedContact(null);
    }
  }

  function getProfile(profileId: string): Profile | undefined {
    if (profileId === user?.id) {
      return {
        id: user.id,
        username,
        full_name: name,
        bio,
        avatar_url: avatar,
      };
    }

    return profilesById[profileId];
  }

  function Avatar({
    profile,
    size = "md",
  }: {
    profile?: Profile | null;
    size?: "sm" | "md" | "lg";
  }) {
    const dimensions = {
